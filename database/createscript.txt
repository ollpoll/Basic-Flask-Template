CREATE TABLE users (
   userid INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
   firstname TEXT NOT NULL,
   lastname TEXT NOT NULL,
   password TEXT NOT NULL,
   email TEXT NOT NULL,
   permission TEXT DEFAULT 'client' CHECK (permission in ('client','valuator','admin')),
   lastaccess DATE DEFAULT (datetime('now','localtime')),
   status TEXT NULL,
   profilephoto TEXT NULL,
   token TEXT NULL
);

CREATE TABLE messages (
   messageid INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
   messagetext TEXT NULL,
   messageimage TEXT NULL,
   author INTEGER NOT NULL,
   FOREIGN KEY(author) REFERENCES users(userid) ON DELETE CASCADE
);

CREATE TABLE valuators (
   userid INTEGER PRIMARY KEY NOT NULL,
   hireprice REAL DEFAULT 0,
   specialty TEXT NULL,
   location TEXT NULL,
   rating REAL DEFAULT 0 CHECK (rating >= 0 AND rating <= 5),
   FOREIGN KEY(userid) REFERENCES users(userid) ON DELETE CASCADE
);
